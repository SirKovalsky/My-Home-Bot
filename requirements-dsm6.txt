# =============================================================================
#  Зависимости для Python 3.8 — штатный пакет "Python3" в DSM 6.2.
#
#  Зачем отдельный файл: свежие aiogram (>=3.14), requests (>=2.33) и
#  python-dotenv (>=1.1) требуют Python >= 3.10/3.9, а на DSM 6.2 стоит
#  Python 3.8. Здесь зафиксированы последние версии, поддерживающие 3.8.
#
#  Ставится автоматически из deploy/install.sh, но можно и вручную:
#      .venv/bin/pip install -r requirements-dsm6.txt
# =============================================================================

aiogram==3.13.1
# Транзитивные зависимости aiogram 3.13.1, зафиксированы для предсказуемости
# (aiogram и так ограничивает их: aiohttp<3.11, pydantic<2.9, aiofiles<24.2).
aiohttp==3.10.11
aiofiles==23.2.1
magic-filter==1.0.12
pydantic==2.8.2
typing-extensions>=4.7.0,<=5.0

# PySocks (через extras socks) обязателен для схем socks5h://.
requests[socks]==2.32.4

# КРИТИЧНО для DSM 6.2: Python 3.8 в пакете Synology собран с OpenSSL 1.0.2u,
# а urllib3 2.x требует OpenSSL 1.1.1+. Без этого пина requests падает с
# "ImportError: urllib3 v2 only supports OpenSSL 1.1.1+".
urllib3>=1.26,<2

transmission-rpc==7.0.12
python-dotenv==1.0.1

# Обход Cloudflare на трекерах: подделывает TLS/HTTP-фингерпринт браузера,
# иначе Cloudflare отдаёт страницу «Just a moment...» (403).
# Есть готовый wheel под Python 3.8: cp38-abi3-manylinux2014_x86_64.
curl_cffi==0.9.0
